package main
import (
 "fmt"
 "strings"
 "time"
 "github.com/keshon/gektor/pkg/tui/components/markdown"
)
func main() {
 var code strings.Builder; for i:=0;i<200;i++ { fmt.Fprintf(&code,"const value%d=(a,b)=>({sum:a+b, product:a*b});\n",i) }; body:=code.String()
 for _, count:=range []int{1,4,4} {
  src:=strings.Repeat("~~~javascript\n"+body+"~~~\n\n",count)
  start:=time.Now(); rows:=markdown.Render(src+"Streaming suffix",100,markdown.Styles{})
  fmt.Printf("closed=%d input_bytes=%d output_rows=%d elapsed=%s\n",count,len(src),len(rows),time.Since(start))
 }
}
