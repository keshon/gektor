package agent

import (
	"path/filepath"
	"testing"
	"time"

	"github.com/keshon/gektor/internal/llm"
)

func TestAuditBudgetPreservesContext(t *testing.T) {
	client := &budgetClient{}
	stateFile := filepath.Join(t.TempDir(), "state.json")
	a := New(Config{
		Client:     client,
		Tools:      NewRegistry(),
		System:     "system sentinel",
		StateFile:  stateFile,
		MaxSteps:   4,
		RunTimeout: time.Second,
	})
	_, _ = a.Run(t.Context(), "task sentinel")
	if len(client.closing.Messages) < 2 || client.closing.Messages[0].Role != llm.RoleSystem {
		t.Errorf("closing request lost context: %#v", client.closing.Messages)
	}
	if a.snapshot.History[0].Role != llm.RoleSystem || a.snapshot.History[0].Content != "system sentinel" {
		t.Errorf("archive root corrupted: %#v", a.snapshot.History[0])
	}

	history, err := LoadState(stateFile)
	if err != nil {
		t.Fatal(err)
	}
	if len(history) == 0 || history[0].Role != llm.RoleSystem || history[0].Content != "system sentinel" {
		t.Errorf("persisted context lost its system message: %#v", history)
	}
}
