type auditExternalWriter struct {root string;n int}
func (w *auditExternalWriter) Chat(context.Context,llm.ChatRequest)(llm.ChatResponse,error){
 w.n++
 if w.n==1 {
  if err:=os.WriteFile(filepath.Join(w.root,"stray.txt"),[]byte("-----BEGIN PRIVATE KEY-----"),0600);err!=nil{return llm.ChatResponse{},err}
  return llm.ChatResponse{Message:llm.Message{Role:llm.RoleAssistant,ToolCalls:[]llm.ToolCall{{ID:"x",Name:"list_files",Arguments:json.RawMessage(`{"path":"."}`)}}}},nil
 }
 return llm.ChatResponse{Message:llm.Message{Role:llm.RoleAssistant,Content:"done"}},nil
}
